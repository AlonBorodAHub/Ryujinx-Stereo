cmake_minimum_required(VERSION 3.24)
project(ZeldaLeiaBridge LANGUAGES CXX)
set(LEIASR_SDKROOT "" CACHE PATH "Installed LeiaSR SDK")
if(NOT LEIASR_SDKROOT)
  message(FATAL_ERROR "Pass -DLEIASR_SDKROOT=<installed LeiaSR SDK folder>")
endif()
list(APPEND CMAKE_PREFIX_PATH "${LEIASR_SDKROOT}/lib/cmake")
find_package(simulatedreality REQUIRED)
find_package(srOpenGL REQUIRED)
add_library(ZeldaLeiaBridge SHARED bridge.cpp)
target_compile_features(ZeldaLeiaBridge PRIVATE cxx_std_20)
target_compile_definitions(ZeldaLeiaBridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(ZeldaLeiaBridge PRIVATE opengl32 simulatedreality srOpenGL::srOpenGL)
